def make_api_request(self, keywords):
    """Make API request to news sources with keywords."""
    api_key = os.environ.get("NEWS_API_KEY")
    if not api_key:
        raise ValueError("NEWS_API_KEY environment variable not set")
        
    # Construct query from keywords with weighting
    query = " OR ".join([k for k in keywords])
    
    # Make request with proper headers and parameters
    url = "https://newsapi.org/v2/everything"
    params = {
        "q": query,
        "sortBy": "relevancy",
        "language": "en",
        "pageSize": 25,
        "apiKey": api_key
    }
    
    headers = {
        "User-Agent": "NewsToVideo/1.0"
    }
    
    return requests.get(url, params=params, headers=headers)
